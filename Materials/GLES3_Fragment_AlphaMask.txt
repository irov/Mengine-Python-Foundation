#version 300 es

uniform sampler2D inSampler0;
uniform sampler2D inSampler1;

in highp vec2 v_UV0;
in highp vec2 v_UV1;
in mediump vec4 v_Col;

layout(location = 0) out mediump vec4 outColor;

void main(void)
{
	mediump vec4 c0 = texture(inSampler0, v_UV0);
    mediump vec4 c1 = texture(inSampler1, v_UV1);

    mediump vec4 c = vec4(v_Col.xyz * c0.xyz, v_Col.a * c1.a);

	outColor = c;
}
