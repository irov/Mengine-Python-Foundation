#version 300 es

uniform sampler2D inSampler0;
uniform mediump float u_factor;
uniform mediump vec4 u_color;

in mediump vec4 v_Col;
in highp vec2 v_UV0;

layout(location = 0) out mediump vec4 outColor;

void main(void)
{
    mediump vec4 color = texture(inSampler0, v_UV0) * vec4(u_color.xyz, 1.0);
    mediump float l = color.x * 0.3 + color.y * 0.59 + color.z * 0.11;
    mediump vec4 d = vec4(l, l, l, color.w);

    outColor = v_Col * mix(color, d, u_factor);
}
