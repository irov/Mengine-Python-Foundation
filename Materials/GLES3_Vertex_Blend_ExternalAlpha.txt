#version 300 es

uniform highp mat4 wvpMatrix;

in highp vec4 inVert;
in mediump vec4 inCol;
in highp vec2 inUV0;
in highp vec2 inUV1;

out mediump vec4 v_Col;
out highp vec2 v_UV0;
out highp vec2 v_UV1;

void main(void)
{
    gl_Position = wvpMatrix * inVert;
    v_Col = inCol;
    v_UV0 = inUV0;
    v_UV1 = inUV1;
}
